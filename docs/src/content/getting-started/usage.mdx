### Create the store

<If adapterId="react-grid">

Create the grid store with `useGridStore` and your grid data, then read the active schema using `useGridStoreSelector`:

```typescript
import { basicJSON } from "@jsoc/grid-examples-core";

const gridStore = useGridStore(basicJSON);
const activeSchema = useGridStoreSelector(gridStore, (store) =>
  store.getActiveSchema(),
);
```

</If>

<If adapterId="vanilla-grid">

Create the grid store with `createGridStore` and your grid data, then read the active schema config:

```typescript
import { basicJSON } from "@jsoc/grid-examples-core";

const gridStore = createGridStore({ data: basicJSON });
const gridOptions = gridStore.getActiveSchema().config;
```

</If>

<If adapterId="vue-grid">

Create the grid store with `useGridStore` and your grid data, then read the active schema using `useGridStoreSelector`:

```typescript
import { basicJSON } from "@jsoc/grid-examples-core";

const gridStore = useGridStore(basicJSON);
const activeSchema = useGridStoreSelector(gridStore, (store) =>
  store.getActiveSchema(),
);
```

</If>

### Render grid component

Pass the generated config to the grid component:

<If adapterId="react-grid" pluginId="ag">

```tsx
<AgGridReact key={activeSchema.id} {...activeSchema.config} />
```

</If>

<If adapterId="react-grid" pluginId="ant">

```tsx
<Table key={activeSchema.id} {...activeSchema.config} />
```

</If>

<If adapterId="react-grid" pluginId="mantine">

```tsx
const table = useMantineReactTable(activeSchema.config);

<MantineReactTable table={table} />
```

</If>

<If adapterId="react-grid" pluginId="mui">

```tsx
<DataGrid key={activeSchema.id} {...activeSchema.config} />
```

</If>

<If adapterId="react-grid" pluginId="prime">

```tsx
const { columns, ...config } = activeSchema.config;

<DataTable key={activeSchema.id} {...config}>
  {columns.map((col) => (
    <Column key={col.field} {...col} />
  ))}
</DataTable>
```

</If>

<If adapterId="react-grid" pluginId="tanstack">

```tsx
const table = useReactTable({
  ...activeSchema.config,
  getCoreRowModel: getCoreRowModel(),
});

return (
  <table>
    <thead>
      {table.getHeaderGroups().map((headerGroup) => (
        <tr key={headerGroup.id}>
          {headerGroup.headers.map((header) => (
            <th key={header.id}>
              {flexRender(header.column.columnDef.header, header.getContext())}
            </th>
          ))}
        </tr>
      ))}
    </thead>
    <tbody>
      {table.getRowModel().rows.map((row) => (
        <tr key={row.id}>
          {row.getVisibleCells().map((cell) => (
            <td key={cell.id}>
              {flexRender(cell.column.columnDef.cell, cell.getContext())}
            </td>
          ))}
        </tr>
      ))}
    </tbody>
  </table>
);
```

</If>

<If adapterId="vanilla-grid" pluginId="ag">

</If>

<If adapterId="vue-grid" pluginId="ag">

```vue

  <AgGridVue :key="activeSchema.id" :grid-options="activeSchema.config" />

```

</If>

<If adapterId="vue-grid" pluginId="tanstack">

```vue
<script setup lang="ts">
const table = useVueTable({
  ...activeSchema.config,
  getCoreRowModel: getCoreRowModel(),
});
</script>

<template>
  <table>
    <thead>
      <tr v-for="headerGroup in table.getHeaderGroups()" :key="headerGroup.id">
        <th v-for="header in headerGroup.headers" :key="header.id">
          <FlexRender
            v-if="!header.isPlaceholder"
            :render="header.column.columnDef.header"
            :props="header.getContext()"
          />
        </th>
      </tr>
    </thead>
    <tbody>
      <tr v-for="row in table.getRowModel().rows" :key="row.id">
        <td v-for="cell in row.getVisibleCells()" :key="cell.id">
          <FlexRender
            :render="cell.column.columnDef.cell"
            :props="cell.getContext()"
          />
        </td>
      </tr>
    </tbody>
  </table>
</template>
```

</If>
